"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { Layers, Plus, Calendar, Users, Target, MoreVertical, Loader2, X } from "lucide-react";

type Cohort = {
  id: string;
  name: string;
  status: 'draft' | 'active' | 'completed';
  start_date: string;
  end_date: string;
};

export default function CohortesPage() {
  const [cohorts, setCohorts] = useState<Cohort[]>([]);
  const [loading, setLoading] = useState(true);
  
  // États pour la modale de création
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [saving, setSaving] = useState(false);
  const [formData, setFormData] = useState({
    name: "",
    status: "draft" as Cohort['status'],
    start_date: "",
    end_date: ""
  });

  // Charger les données depuis Supabase
  useEffect(() => {
    fetchCohorts();
  }, []);

  const fetchCohorts = async () => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from("cohorts")
        .select("*")
        .order("start_date", { ascending: false });

      if (error) throw error;
      if (data) setCohorts(data);
    } catch (error: any) {
      alert("❌ Erreur de chargement : " + error.message);
    } finally {
      setLoading(false);
    }
  };

  // Créer un nouveau programme
  const handleCreateCohort = async (e: React.FormEvent) => {
    e.preventDefault();
    
    if (new Date(formData.start_date) >= new Date(formData.end_date)) {
      alert("❌ La date de fin doit être après la date de début !");
      return;
    }

    setSaving(true);
    try {
      const { data, error } = await supabase
        .from("cohorts")
        .insert([formData])
        .select();

      if (error) throw error;

      if (data) {
        setCohorts([data[0], ...cohorts]); // Ajoute le nouveau programme visuellement
        setIsModalOpen(false); // Ferme la modale
        setFormData({ name: "", status: "draft", start_date: "", end_date: "" }); // Reset
        alert("✅ Programme créé avec succès !");
      }
    } catch (error: any) {
      alert("❌ Erreur de création : " + error.message);
    } finally {
      setSaving(false);
    }
  };

  // Fonction intelligente pour calculer la progression en temps réel
  const calculateProgress = (start: string, end: string) => {
    const startDate = new Date(start).getTime();
    const endDate = new Date(end).getTime();
    const now = new Date().getTime();

    if (now < startDate) return 0; // Pas encore commencé
    if (now > endDate) return 100; // Terminé

    const totalDuration = endDate - startDate;
    const elapsed = now - startDate;
    return Math.round((elapsed / totalDuration) * 100);
  };

  // Traduction des statuts pour l'affichage
  const statusConfig = {
    draft: { label: "Brouillon / Inscriptions", color: "bg-gray-100 text-gray-600" },
    active: { label: "En cours", color: "bg-blue-50 text-[#2563EB]" },
    completed: { label: "Terminé", color: "bg-emerald-50 text-emerald-600" },
  };

  return (
    <div className="max-w-7xl mx-auto pb-12">
      
      {/* HEADER */}
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-10 gap-4">
        <div>
          <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Organisation</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Programmes & Cohortes</h2>
        </div>
        <button 
          onClick={() => setIsModalOpen(true)}
          className="flex items-center gap-2 bg-[#101D42] hover:bg-[#1a2b5e] text-white px-6 py-3 rounded-full text-xs font-semibold tracking-wide transition-all shadow-sm"
        >
          <Plus size={16} /> CRÉER UN PROGRAMME
        </button>
      </div>

      {/* LISTE DES COHORTES */}
      {loading ? (
        <div className="flex justify-center py-20 text-[#2563EB]"><Loader2 className="animate-spin" size={32} /></div>
      ) : cohorts.length === 0 ? (
        <div className="bg-white rounded-3xl border border-gray-200 p-16 text-center text-[#9CA3AF] shadow-sm">
          <Layers size={48} className="mx-auto mb-4 text-gray-300" />
          <p className="text-lg font-medium text-[#101D42]">Aucun programme actif.</p>
          <p className="text-sm mt-1">Créez votre première cohorte pour y regrouper vos startups.</p>
        </div>
      ) : (
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
          {cohorts.map((cohort) => {
            const progress = calculateProgress(cohort.start_date, cohort.end_date);
            const style = statusConfig[cohort.status];

            return (
              <div key={cohort.id} className="bg-white rounded-[24px] border border-gray-200 p-8 shadow-sm flex flex-col justify-between hover:border-[#2563EB]/30 transition-all group">
                <div>
                  <div className="flex justify-between items-start mb-6">
                    <span className={`text-[10px] font-bold tracking-[0.1em] uppercase px-3 py-1 rounded-full ${style.color}`}>
                      {style.label}
                    </span>
                    <button className="text-gray-400 hover:text-[#101D42]"><MoreVertical size={18} /></button>
                  </div>
                  <h3 className="text-xl font-medium text-[#101D42] mb-4">{cohort.name}</h3>
                  
                  <div className="space-y-3 mb-8">
                    <div className="flex items-center gap-3 text-sm text-[#9CA3AF] font-medium">
                      <Calendar size={16} className="text-[#2563EB]" /> 
                      {new Date(cohort.start_date).toLocaleDateString('fr-FR')} - {new Date(cohort.end_date).toLocaleDateString('fr-FR')}
                    </div>
                    <div className="flex items-center gap-3 text-sm text-[#9CA3AF] font-medium opacity-50">
                      {/* Pour l'instant, on laisse 0. On fera le matching plus tard */}
                      <Users size={16} /> 0 Startups assignées
                    </div>
                  </div>
                </div>

                <div>
                  <div className="flex justify-between text-[10px] font-bold text-[#9CA3AF] uppercase mb-2">
                    <span>Progression du programme</span>
                    <span className="text-[#101D42]">{progress}%</span>
                  </div>
                  <div className="w-full bg-[#F8FAFC] rounded-full h-2">
                    <div className={`h-2 rounded-full ${cohort.status === 'completed' ? 'bg-emerald-500' : 'bg-[#2563EB]'}`} style={{ width: `${progress}%` }}></div>
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {/* ========================================= */}
      {/* MODALE DE CRÉATION                        */}
      {/* ========================================= */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-[#101D42]/40 backdrop-blur-sm">
          <div className="bg-white w-full max-w-md rounded-[32px] overflow-hidden shadow-2xl animate-in fade-in zoom-in-95 duration-200">
            
            <div className="p-6 border-b border-gray-100 flex items-center justify-between bg-[#FAFAFA]">
              <div className="flex items-center gap-3">
                <div className="p-2 bg-blue-50 text-[#2563EB] rounded-lg"><Layers size={20} /></div>
                <h3 className="text-lg font-bold text-[#101D42]">Nouveau Programme</h3>
              </div>
              <button onClick={() => setIsModalOpen(false)} className="text-gray-400 hover:text-[#101D42]"><X size={20} /></button>
            </div>

            <form onSubmit={handleCreateCohort} className="p-8 space-y-6">
              
              <div>
                <label className="block text-xs font-bold tracking-[0.1em] text-[#101D42] uppercase mb-2">Nom de la cohorte <span className="text-red-500">*</span></label>
                <input 
                  type="text" required 
                  value={formData.name} onChange={(e) => setFormData({...formData, name: e.target.value})}
                  className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" 
                  placeholder="Ex: Batch Été 2026" 
                />
              </div>

              <div>
                <label className="block text-xs font-bold tracking-[0.1em] text-[#101D42] uppercase mb-2">Statut initial <span className="text-red-500">*</span></label>
                <select 
                  required 
                  value={formData.status} onChange={(e) => setFormData({...formData, status: e.target.value as Cohort['status']})}
                  className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42] bg-white"
                >
                  <option value="draft">Brouillon (Ouvert aux inscriptions)</option>
                  <option value="active">Actif (En cours)</option>
                </select>
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold tracking-[0.1em] text-[#101D42] uppercase mb-2">Date de début <span className="text-red-500">*</span></label>
                  <input 
                    type="date" required 
                    value={formData.start_date} onChange={(e) => setFormData({...formData, start_date: e.target.value})}
                    className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" 
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold tracking-[0.1em] text-[#101D42] uppercase mb-2">Date de fin <span className="text-red-500">*</span></label>
                  <input 
                    type="date" required 
                    value={formData.end_date} onChange={(e) => setFormData({...formData, end_date: e.target.value})}
                    className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" 
                  />
                </div>
              </div>

              <div className="pt-4">
                <button type="submit" disabled={saving} className="w-full py-4 rounded-full font-bold tracking-wide text-white transition-colors text-xs flex items-center justify-center gap-2 bg-[#101D42] hover:bg-[#1a2b5e]">
                  {saving ? <Loader2 size={16} className="animate-spin" /> : "GÉNÉRER LE PROGRAMME"}

"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { CalendarDays, Plus, MapPin, Video, Users, Loader2, X, MoreVertical } from "lucide-react";

type EventType = {
  id: string;
  title: string;
  type: string;
  event_date: string;
  event_time: string;
  mode: string;
  location: string;
  attendees: number;
};

export default function AdminEventsPage() {
  const [events, setEvents] = useState<EventType[]>([]);
  const [loading, setLoading] = useState(true);
  
  // États pour la modale
  const [isModalOpen, setIsModalOpen] = useState(false);
  const [saving, setSaving] = useState(false);
  
  const [formData, setFormData] = useState({
    title: "",
    type: "Masterclass",
    event_date: "",
    event_time: "",
    mode: "En Ligne",
    location: ""
  });

  // Charger les événements au démarrage
  useEffect(() => {
    fetchEvents();
  }, []);

  const fetchEvents = async () => {
    try {
      setLoading(true);
      const { data, error } = await supabase
        .from("events")
        .select("*")
        .order("event_date", { ascending: true }); // Du plus proche au plus lointain

      if (error) throw error;
      if (data) setEvents(data);
    } catch (error: any) {
      alert("❌ Erreur de chargement : " + error.message);
    } finally {
      setLoading(false);
    }
  };

  const handleCreateEvent = async (e: React.FormEvent) => {
    e.preventDefault();
    setSaving(true);
    
    try {
      const { data, error } = await supabase
        .from("events")
        .insert([{
          title: formData.title,
          type: formData.type,
          event_date: formData.event_date,
          event_time: formData.event_time,
          mode: formData.mode,
          location: formData.location,
          attendees: 0 // Par défaut à la création
        }])
        .select();

      if (error) throw error;

      if (data) {
        // Ajout du nouvel événement et tri
        const newEvents = [...events, data[0]].sort((a, b) => new Date(a.event_date).getTime() - new Date(b.event_date).getTime());
        setEvents(newEvents);
        
        setIsModalOpen(false);
        setFormData({ title: "", type: "Masterclass", event_date: "", event_time: "", mode: "En Ligne", location: "" });
        alert("✅ Événement planifié avec succès !");
      }
    } catch (error: any) {
      alert("❌ Erreur de création : " + error.message);
    } finally {
      setSaving(false);
    }
  };

  // Fonction pour extraire le Jour et le Mois d'une date (Ex: "15 Jan")
  const formatDateForCard = (dateString: string) => {
    const dateObj = new Date(dateString);
    const day = dateObj.toLocaleDateString('fr-FR', { day: '2-digit' });
    const month = dateObj.toLocaleDateString('fr-FR', { month: 'short' }).replace('.', '');
    return { day, month };
  };

  return (
    <div className="max-w-6xl mx-auto pb-12">
      
      {/* HEADER */}
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center mb-10 gap-4">
        <div>
          <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Calendrier</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Gestion des Événements</h2>
        </div>
        <button 
          onClick={() => setIsModalOpen(true)}
          className="flex items-center gap-2 bg-[#101D42] hover:bg-[#1a2b5e] text-white px-6 py-3 rounded-full text-xs font-semibold tracking-wide transition-all shadow-sm"
        >
          <Plus size={16} /> PLANIFIER UN ÉVÉNEMENT
        </button>
      </div>

      {/* LISTE DES ÉVÉNEMENTS */}
      {loading ? (
        <div className="flex justify-center py-20 text-[#2563EB]"><Loader2 className="animate-spin" size={32} /></div>
      ) : events.length === 0 ? (
        <div className="bg-white rounded-3xl border border-gray-200 p-16 text-center text-[#9CA3AF] shadow-sm">
          <CalendarDays size={48} className="mx-auto mb-4 text-gray-300" />
          <p className="text-lg font-medium text-[#101D42]">Le calendrier est vide.</p>
          <p className="text-sm mt-1">Planifiez votre premier événement pour l'écosystème.</p>
        </div>
      ) : (
        <div className="space-y-6">
          {events.map((event) => {
            const { day, month } = formatDateForCard(event.event_date);
            
            return (
              <div key={event.id} className="bg-white rounded-[24px] border border-gray-200 p-6 shadow-sm flex flex-col md:flex-row gap-6 items-center hover:border-[#2563EB]/40 transition-colors group relative">
                
                {/* Bouton Options (en haut à droite) */}

                  <MoreVertical size={18} />
                </button>

                <div className="bg-[#F8FAFC] border border-gray-100 rounded-2xl w-24 h-24 flex flex-col items-center justify-center shrink-0 group-hover:bg-[#101D42] group-hover:text-white transition-colors">

"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { 
  CheckCircle2, XCircle, ExternalLink, Phone, Loader2, Search, 
  AlertTriangle, PauseCircle, Eye, FileText, Download, FolderOpen, Globe
} from "lucide-react";

type ValidStatus = "pending" | "under_review" | "approved" | "rejected" | "suspended";

type Profile = {
  id: string;
  full_name: string;
  role: string;
  status: ValidStatus;
  phone: string;
  company: string;
  application_details: any;
  rejection_reason: string;
  created_at: string;
  startup_profiles?: any[];
  mentor_profiles?: any[];
  investor_profiles?: any[];
  kyc_documents?: any[];
};

export default function AdminUsersPage() {
  const [adminId, setAdminId] = useState<string | null>(null);
  const [candidates, setCandidates] = useState<Profile[]>([]);
  const [loading, setLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState<string | null>(null);
  const [activeTab, setActiveTab] = useState<ValidStatus>("pending");
  const [searchQuery, setSearchQuery] = useState("");

  const [modalOpen, setModalOpen] = useState(false);
  const [selectedCandidate, setSelectedCandidate] = useState<Profile | null>(null);
  const [decisionType, setDecisionType] = useState<ValidStatus | null>(null);
  const [rejectionReason, setRejectionReason] = useState("");
  const [internalComment, setInternalComment] = useState("");

  useEffect(() => {
    fetchData();
  }, []);

  const fetchData = async () => {
    try {
      setLoading(true);
      const { data: { user } } = await supabase.auth.getUser();
      if (user) setAdminId(user.id);

      const { data, error } = await supabase
        .from("profiles")
        .select("*, startup_profiles(*), mentor_profiles(*), investor_profiles(*), kyc_documents(*)")
        .neq("role", "admin")
        .order("created_at", { ascending: false });

      if (error) throw error;
      if (data) setCandidates(data as Profile[]);
    } catch (error: any) {
      alert("❌ Erreur de chargement : " + error.message);
    } finally {
      setLoading(false);
    }
  };

  const startReview = async (candidate: Profile) => {
    try {
      setActionLoading(candidate.id);
      const { error: updateError } = await supabase.from("profiles").update({ status: "under_review", reviewed_by: adminId, reviewed_at: new Date().toISOString() }).eq("id", candidate.id);
      if (updateError) throw updateError;

      await supabase.from("review_logs").insert([{ profile_id: candidate.id, admin_id: adminId, previous_status: candidate.status, new_status: "under_review", comment: "Prise en charge du dossier" }]);
      setCandidates(candidates.map(c => c.id === candidate.id ? { ...c, status: "under_review" } : c));
    } catch (error: any) {
      alert("❌ Erreur : " + error.message);
    } finally {
      setActionLoading(null);
    }
  };

  const handleDecision = async (id: string, newStatus: ValidStatus, name: string) => {
    try {
      setActionLoading(id);
      const { error } = await supabase.from("profiles").update({ status: newStatus }).eq("id", id);
      if (error) throw error;
      setCandidates(candidates.map(c => c.id === id ? { ...c, status: newStatus } : c));
    } catch (error: any) {
      alert("❌ Erreur : " + error.message);
    } finally {
      setActionLoading(null);
    }
  };

  const openDecisionModal = (candidate: Profile, type: ValidStatus) => {
    setSelectedCandidate(candidate);
    setDecisionType(type);
    setRejectionReason(candidate.rejection_reason || "");
    setInternalComment("");
    setModalOpen(true);
  };

  const handleDecisionSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!selectedCandidate || !decisionType || !adminId) return;

    try {
      setActionLoading(selectedCandidate.id);
      const updateData: any = { status: decisionType, reviewed_by: adminId, reviewed_at: new Date().toISOString(), rejection_reason: decisionType === 'rejected' ? rejectionReason : null };
      
      const { error: updateError } = await supabase.from("profiles").update(updateData).eq("id", selectedCandidate.id);
      if (updateError) throw updateError;

      await supabase.from("review_logs").insert([{ profile_id: selectedCandidate.id, admin_id: adminId, previous_status: selectedCandidate.status, new_status: decisionType, comment: internalComment || `Passage au statut ${decisionType}` }]);
      setCandidates(candidates.map(c => c.id === selectedCandidate.id ? { ...c, ...updateData } : c));
      setModalOpen(false);
    } catch (error: any) {
      alert("❌ Erreur : " + error.message);
    } finally {
      setActionLoading(null);
    }
  };

  // NOUVEAU : LA VRAIE FONCTION DE TÉLÉCHARGEMENT SECURISE
  const handleDownloadDoc = async (filePath: string) => {
    if (!filePath) return;
    try {
      // Demande une URL signée valable 60 secondes à Supabase
      const { data, error } = await supabase.storage.from('kyc_files').createSignedUrl(filePath, 60);
      if (error) throw error;
      
      if (data?.signedUrl) {
        // Ouvre le PDF téléchargé dans un nouvel onglet
        window.open(data.signedUrl, '_blank');
      }
    } catch (error: any) {
      alert("❌ Erreur d'accès au document : " + error.message);
    }
  };

  const getDetails = (c: Profile) => {
    let d = c.role === 'startup' ? c.startup_profiles : c.role === 'mentor' ? c.mentor_profiles : c.investor_profiles;
    if (Array.isArray(d)) return d[0] || {}; 
    return d || {}; 
  };

  const filteredCandidates = candidates.filter(c => {
    const details = getDetails(c);
    const companyName = c.company || details.startup_name || details.fund_name || "";
    return c.status === activeTab && (
      c.full_name.toLowerCase().includes(searchQuery.toLowerCase()) || 
      companyName.toLowerCase().includes(searchQuery.toLowerCase())
    );
  });

  return (
    <div className="max-w-7xl mx-auto pb-12">
      <div className="mb-8">
        <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Modération & Rôles</p>
        <h2 className="text-2xl font-medium text-[#101D42]">Utilisateurs & Matching</h2>
      </div>

      <div className="flex flex-col md:flex-row justify-between items-start md:items-center gap-4 mb-6 border-b border-gray-200 pb-4">
        <div className="flex gap-4 text-sm font-medium overflow-x-auto w-full md:w-auto hide-scrollbar">
          {[
            { id: "pending", label: "Nouveaux" }, { id: "under_review", label: "En Examen" },
            { id: "approved", label: "Approuvés" }, { id: "rejected", label: "Refusés" }, { id: "suspended", label: "Suspendus" }
          ].map(tab => (
            <button key={tab.id} onClick={() => setActiveTab(tab.id as ValidStatus)} className={`pb-2 px-1 whitespace-nowrap transition-colors relative ${activeTab === tab.id ? "text-[#101D42] border-b-2 border-[#101D42] font-bold" : "text-[#9CA3AF] hover:text-[#101D42]"}`}>
              {tab.label} <span className="text-xs bg-gray-100 px-2 py-0.5 rounded-full ml-1">{candidates.filter(c => c.status === tab.id).length}</span>
            </button>
          ))}
        </div>
        <div className="relative w-full md:w-64">
          <Search size={14} className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400" />
          <input type="text" placeholder="Rechercher..." value={searchQuery} onChange={(e) => setSearchQuery(e.target.value)} className="w-full pl-9 pr-4 py-2 bg-white border border-gray-200 rounded-full text-sm focus:outline-none focus:border-[#2563EB]" />
        </div>
      </div>

      {loading ? (
        <div className="flex justify-center py-20 text-[#2563EB]"><Loader2 className="animate-spin" size={32} /></div>
      ) : filteredCandidates.length === 0 ? (
        <div className="bg-white rounded-3xl border border-gray-200 p-16 text-center text-[#9CA3AF] shadow-sm"><CheckCircle2 size={48} className="mx-auto mb-4 text-gray-300" /><p className="text-lg font-medium text-[#101D42]">Aucun dossier ici.</p></div>
      ) : (
        <div className="space-y-6">
          {filteredCandidates.map((candidate) => {
            const details = getDetails(candidate);
            const displayCompany = candidate.company || details.startup_name || details.fund_name || "Organisation non spécifiée";

            return (
              <div key={candidate.id} className="bg-white rounded-[24px] border border-gray-200 shadow-sm flex flex-col transition-all hover:border-[#2563EB]/30 overflow-hidden">
                
                <div className="p-6 lg:p-8 flex flex-col lg:flex-row gap-8">
                  
                  {/* Infos principales */}
                  <div className="flex-1 space-y-4">
                    <div className="flex items-center gap-3 mb-2">
                      <span className={`text-[10px] font-bold tracking-[0.1em] uppercase px-3 py-1 rounded-full ${candidate.role === 'startup' ? 'bg-blue-50 text-[#2563EB]' : candidate.role === 'mentor' ? 'bg-emerald-50 text-emerald-600' : 'bg-orange-50 text-orange-600'}`}>
                        {candidate.role}
                      </span>
                      <span className="text-[10px] font-bold uppercase tracking-wider text-gray-400">
                        {new Date(candidate.created_at).toLocaleDateString('fr-FR')}
                      </span>
                    </div>

                    <div>
                      <h3 className="text-xl font-medium text-[#101D42]">{candidate.full_name}</h3>
                      <p className="text-sm text-[#9CA3AF] font-medium flex items-center gap-2 mt-1">
                        {displayCompany}
                        {candidate.phone && <><span className="w-1 h-1 bg-gray-300 rounded-full"></span> <Phone size={12} /> {candidate.phone}</>}
                      </p>
                    </div>

                    <div className="bg-[#F8FAFC] p-5 rounded-2xl text-sm space-y-3 text-gray-600 border border-gray-100 min-h-[80px]">
                      {(!details || Object.keys(details).length === 0) ? (
                        <p className="text-xs text-gray-400 italic flex items-center gap-2 h-full">
                          <AlertTriangle size={14} className="text-orange-400" /> Ce candidat n'a pas encore complété son dossier.
                        </p>
                      ) : (
                        <>
                          {details.description && candidate.role === 'startup' && <p className="italic text-[#101D42]">"{details.description}"</p>}
                          {details.sector && candidate.role === 'startup' && <p><strong>Secteur :</strong> {details.sector}</p>}
                          
                          {details.expertise_domain && <p><strong>Expertise :</strong> {details.expertise_domain}</p>}
                          {details.years_experience !== undefined && <p><strong>Expérience :</strong> {details.years_experience} ans</p>}
                          {details.bio && <p className="italic">"{details.bio}"</p>}
                          {details.availability && <p><strong>Disponibilité :</strong> {details.availability}</p>}
                          
                          {details.target_sectors && <p><strong>Cibles :</strong> {details.target_sectors}</p>}
                          {details.ticket_size && <p><strong>Ticket cible :</strong> {details.ticket_size}</p>}
                          {details.investment_thesis && <p className="italic">"{details.investment_thesis}"</p>}
                          
                          {details.linkedin_url && (
                            <a href={details.linkedin_url.startsWith('http') ? details.linkedin_url : `https://${details.linkedin_url}`} target="_blank" rel="noreferrer" className="text-[#2563EB] inline-flex items-center gap-1.5 hover:underline font-medium mt-2">
                              <ExternalLink size={14} /> Profil LinkedIn
                            </a>
                          )}
                          {details.website && (
                            <a href={details.website.startsWith('http') ? details.website : `https://${details.website}`} target="_blank" rel="noreferrer" className="text-[#101D42] inline-flex items-center gap-1.5 hover:underline font-medium mt-2 ml-4">
                              <Globe size={14} /> Site Web
                            </a>
                          )}
                        </>
                      )}
                    </div>
                  </div>

                  {/* DOCUMENTS JOINTS */}

                            <div className="flex items-center gap-3 overflow-hidden">
