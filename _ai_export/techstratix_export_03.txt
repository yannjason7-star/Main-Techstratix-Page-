import { supabase } from "@/lib/supabase";

import Link from "next/link";
import { useRouter } from "next/navigation";
import { motion, AnimatePresence } from "framer-motion";

// Les 3 messages qui vont tourner en boucle
const SLIDER_MESSAGES = [
  {
    title: "Accélérez votre croissance.",
    desc: "Un écosystème conçu pour structurer votre projet, monter en compétences et réussir votre levée de fonds.",
    icon: <Rocket size={32} className="text-[#2563EB]" />
  },
  {
    title: "Transmettez votre expertise.",
    desc: "Accompagnez des fondateurs ambitieux et participez activement à l'innovation de demain.",
    icon: <Users size={32} className="text-emerald-400" />
  },
  {
    title: "Découvrez les pépites tech.",
    desc: "Un deal flow exclusif, qualifié et organisé pour vos prochains investissements stratégiques.",
    icon: <TrendingUp size={32} className="text-orange-400" />
  }
];

export default function LoginPage() {
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [loading, setLoading] = useState(false);
  
  const [errorType, setErrorType] = useState<"credentials" | "pending" | "under_review" | "rejected" | "suspended" | null>(null);
  const [rejectionReason, setRejectionReason] = useState("");

  // Gestion du Carousel Animé
  const [currentIndex, setCurrentIndex] = useState(0);
  useEffect(() => {
    const timer = setInterval(() => {
      setCurrentIndex((prev) => (prev + 1) % SLIDER_MESSAGES.length);
    }, 5000); // Change le texte toutes les 5 secondes
    return () => clearInterval(timer);
  }, []);

  const handleLogin = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    setErrorType(null);

    try {
      const { data: authData, error: authError } = await supabase.auth.signInWithPassword({ email, password });
      
      if (authError || !authData.user) {
        setErrorType("credentials");
        setLoading(false);
        return;
      }

      const { data: profile, error: profileError } = await supabase
        .from('profiles')
        .select('role, status, rejection_reason')
        .eq('id', authData.user.id)
        .single();

      if (profileError || !profile) throw new Error("Profil introuvable.");

      switch (profile.status) {
        case 'approved':
          if (profile.role === 'admin') router.push("/admin");
          else router.push("/dashboard");
          break;
        case 'pending':
          router.push("/onboarding");
          break;
        case 'under_review':
          await supabase.auth.signOut();
          setErrorType("under_review");
          break;
        case 'rejected':
          await supabase.auth.signOut();
          setRejectionReason(profile.rejection_reason || "Non spécifié.");
          setErrorType("rejected");
          break;
        case 'suspended':
          await supabase.auth.signOut();
          setErrorType("suspended");
          break;
        default:
          await supabase.auth.signOut();
          setErrorType("credentials");
      }
    } catch (error: any) {
      console.error(error);
      setErrorType("credentials");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen flex bg-[#FAFAFA]">
      
      {/* GAUCHE : Le Formulaire */}
      <div className="w-full lg:w-1/2 flex flex-col justify-center items-center p-8 overflow-y-auto">
        <div className="w-full max-w-md">
          
          <Link href="/" className="inline-flex items-center gap-2 text-sm text-[#9CA3AF] hover:text-[#101D42] mb-10 transition-colors">
            <ArrowLeft size={16} /> Retour à l'accueil
          </Link>

          <h1 className="text-3xl font-medium tracking-tight text-[#101D42] mb-2">Connexion</h1>
          <p className="text-[#9CA3AF] text-sm mb-8">Accédez à votre espace sécurisé.</p>

          <form onSubmit={handleLogin} className="space-y-4">
            <div>
              <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">E-mail</label>
              <input type="email" required value={email} onChange={(e) => setEmail(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="nom@exemple.com" />
            </div>
            <div>
              <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Mot de passe</label>
              <input type="password" required value={password} onChange={(e) => setPassword(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="••••••••" />
            </div>

            {errorType && (
              <div className="p-4 rounded-xl flex items-start gap-3 bg-red-50 border border-red-100 text-red-700 text-sm font-medium">
                <ShieldAlert size={18} className="shrink-0 mt-0.5" />
                <div>
                  {errorType === "credentials" && "Identifiants incorrects ou compte inexistant."}
                  {errorType === "under_review" && "Votre dossier est en cours d'analyse par notre équipe. Vous serez notifié très bientôt."}
                  {errorType === "suspended" && "Votre compte a été suspendu par l'administration."}
                  {errorType === "rejected" && <><p className="font-bold mb-1">Dossier refusé.</p><p className="text-xs text-red-600">Motif : {rejectionReason}</p></>}
                </div>
              </div>
            )}

            <button type="submit" disabled={loading} className="w-full bg-[#101D42] hover:bg-[#1a2b5e] text-white font-semibold py-4 rounded-full tracking-wide transition-colors text-sm disabled:opacity-70 mt-4">
              {loading ? <Loader2 className="animate-spin mx-auto" size={20} /> : "SE CONNECTER"}
            </button>
          </form>

          <div className="mt-8 text-center text-sm">
            <span className="text-[#9CA3AF]">Vous n'avez pas de compte ?</span>{" "}
            <Link href="/register" className="text-[#101D42] font-semibold hover:underline">S'inscrire</Link>
          </div>
        </div>
      </div>

      {/* DROITE : LE PANNEAU MARKETING ANIMÉ */}
      <div className="hidden lg:flex w-1/2 bg-[#060B14] p-16 flex-col justify-between relative overflow-hidden">
        
        {/* Lumières d'arrière-plan */}
        <div className="absolute top-[-10%] right-[-10%] w-[500px] h-[500px] bg-[#2563EB] opacity-20 rounded-full blur-[120px] pointer-events-none"></div>
        <div className="absolute bottom-[-10%] left-[-10%] w-[400px] h-[400px] bg-emerald-500 opacity-10 rounded-full blur-[100px] pointer-events-none"></div>

        {/* Logo en haut */}
        <div className="relative z-10">

            <span className="text-2xl font-medium tracking-tight text-white">TechStratix</span>

        {/* Textes rotatifs (Le Carousel) */}
        <div className="relative z-10 h-64 flex flex-col justify-center">

              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: -20 }}
              transition={{ duration: 0.5, ease: "easeOut" }}

              <div className="mb-6 bg-white/5 w-16 h-16 rounded-2xl flex items-center justify-center border border-white/10 backdrop-blur-sm shadow-xl">

"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { useRouter } from "next/navigation";
import { Loader2, CheckCircle2, Rocket, Users, TrendingUp, Building2, Globe, FileText, AlertTriangle, UploadCloud, MapPin, Clock, Award, ShieldCheck, Briefcase, LogOut } from "lucide-react";import Link from "next/link";
import BorderGlow from "@/components/BorderGlow";
import StarBorder from "@/components/StarBorder";
import { motion } from "framer-motion";

export default function OnboardingPage() {
  const router = useRouter();
  
  const [userId, setUserId] = useState<string | null>(null);
  const [userRole, setUserRole] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [errorMsg, setErrorMsg] = useState("");
  const [isSubmitted, setIsSubmitted] = useState(false);

  // States unifiés pour tous les formulaires
  const [formData, setFormData] = useState({
    name: "", // startup_name, fund_name
    sector: "", // secteut de la startup, ou target_sectors de l'investor
    customSector: "",
    country: "",
    city: "",
    linkedin: "",
    website: "",
    description: "", // pitch, bio, ou investment_thesis
    experience: 0,
    availability: "", 
    ticketSize: "" // Pour l'investisseur
  });

  const [termsAccepted, setTermsAccepted] = useState(false);
  const [shakeTerms, setShakeTerms] = useState(false);

  // Fichiers (CV, Pitch, Justificatifs...)
  const [mainFile, setMainFile] = useState<File | null>(null); 
  const [secondaryFile, setSecondaryFile] = useState<File | null>(null); 
// Fonction pour quitter proprement et reprendre plus tard
  const handleSignOut = async () => {
    if (confirm("Voulez-vous vraiment vous déconnecter ? Vous pourrez reprendre ce formulaire plus tard en vous reconnectant.")) {
      setSaving(true);
      await supabase.auth.signOut();
      router.push("/");
    }
  };
  // Verrouillage de la page
  useEffect(() => {
    const handleBeforeUnload = (e: BeforeUnloadEvent) => {
      if (!isSubmitted && (formData.name || formData.description)) {
        e.preventDefault();
        e.returnValue = '';
      }
    };
    window.addEventListener('beforeunload', handleBeforeUnload);
    return () => window.removeEventListener('beforeunload', handleBeforeUnload);
  }, [formData, isSubmitted]);

  // Initialisation
  useEffect(() => {
    const initOnboarding = async () => {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return router.push("/login");

      const { data: profile } = await supabase.from("profiles").select("role, status").eq("id", user.id).single();
      if (!profile || profile.status === "approved" || profile.status === "under_review" || profile.status === "rejected") {
        return router.push("/dashboard");
      }

      let hasCompletedData = false;
      if (profile.role === 'startup') {
        const { data } = await supabase.from('startup_profiles').select('profile_id').eq('profile_id', user.id).single();
        if (data) hasCompletedData = true;
      } else if (profile.role === 'mentor') {
        const { data } = await supabase.from('mentor_profiles').select('profile_id').eq('profile_id', user.id).single();
        if (data) hasCompletedData = true;
      } else if (profile.role === 'investor') {
        const { data } = await supabase.from('investor_profiles').select('profile_id').eq('profile_id', user.id).single();
        if (data) hasCompletedData = true;
      }

      if (hasCompletedData) {
        setIsSubmitted(true);
        await supabase.auth.signOut();
        setLoading(false);
        return;
      }

      setUserId(user.id);
      setUserRole(profile.role);
      setLoading(false);
    };
    initOnboarding();
  }, [router]);

const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setErrorMsg("");

    if (!termsAccepted) {
      setShakeTerms(true);
      setTimeout(() => setShakeTerms(false), 500);
      return;
    }

    setSaving(true);

    try {
      if (!userId || !userRole) throw new Error("Erreur d'identification.");

      const finalSector = formData.sector === "Autre" ? formData.customSector : formData.sector;

      // ==========================================
      // NOUVEAU : FONCTION D'UPLOAD SÉCURISÉ
      // ==========================================
      const uploadFile = async (file: File, prefix: string) => {
        const fileExt = file.name.split('.').pop();
        // On nettoie le nom du fichier pour éviter les bugs (espaces, accents...)
        const cleanName = file.name.replace(/[^a-zA-Z0-9.\-_]/g, '_');
        // Le chemin : ID_UTILISATEUR / PREFIX_TIMESTAMP_NOMFICHIER
        const filePath = `${userId}/${prefix}_${Date.now()}_${cleanName}`;
        
        const { error: uploadError } = await supabase.storage.from('kyc_files').upload(filePath, file);
        if (uploadError) throw new Error(`Erreur d'upload de ${file.name} : ${uploadError.message}`);
        return filePath; // On retourne l'adresse sécurisée
      };

      let mainFilePath = null;
      let secondaryFilePath = null;

      // On lance les vrais uploads dans Supabase Storage avant de sauvegarder le texte
      if (mainFile) mainFilePath = await uploadFile(mainFile, 'main');
      if (secondaryFile) secondaryFilePath = await uploadFile(secondaryFile, 'secondary');

      // ==========================================
      // SAUVEGARDE EN BASE DE DONNÉES
      // ==========================================
      if (formData.country || formData.city) {
        const { error: profileError } = await supabase.from("profiles")
          .update({ country: formData.country, city: formData.city })
          .eq("id", userId);
        if (profileError) throw profileError;
      }

      if (userRole === "startup") {
        const { error } = await supabase.from("startup_profiles").insert([{
          profile_id: userId, startup_name: formData.name, sector: finalSector, description: formData.description, website: formData.website || null,
          pitch_deck_path: mainFilePath // LE VRAI CHEMIN !
        }]);
        if (error) throw error;
      } 
      else if (userRole === "mentor") {
        const { error } = await supabase.from("mentor_profiles").insert([{
          profile_id: userId, expertise_domain: finalSector, bio: formData.description, years_experience: formData.experience, 
          linkedin_url: formData.linkedin, availability: formData.availability
        }]);
        if (error) throw error;
        
        if (mainFile && mainFilePath) await supabase.from("kyc_documents").insert([{ profile_id: userId, document_type: 'other', storage_path: mainFilePath, file_name: mainFile.name }]);
        if (secondaryFile && secondaryFilePath) await supabase.from("kyc_documents").insert([{ profile_id: userId, document_type: 'other', storage_path: secondaryFilePath, file_name: secondaryFile.name }]);
      } 
      else if (userRole === "investor") {
        const { error } = await supabase.from("investor_profiles").insert([{
          profile_id: userId, fund_name: formData.name, target_sectors: finalSector, ticket_size: formData.ticketSize, 
          linkedin_url: formData.linkedin, investment_thesis: formData.description
        }]);
        if (error) throw error;

        if (mainFile && mainFilePath) await supabase.from("kyc_documents").insert([{ profile_id: userId, document_type: 'id_card', storage_path: mainFilePath, file_name: mainFile.name }]);
      }

      // Déconnexion finale
      await supabase.auth.signOut();
      setIsSubmitted(true);
      window.scrollTo({ top: 0, behavior: 'smooth' });

    } catch (error: any) {
      setErrorMsg("Erreur lors de l'enregistrement : " + error.message);
    } finally {
      setSaving(false);
    }
  };

  if (loading) return <div className="min-h-screen flex items-center justify-center bg-[#FAFAFA]"><Loader2 className="animate-spin text-[#2563EB]" size={40} /></div>;

  if (isSubmitted) {
    return (
      <div className="min-h-screen bg-[#FAFAFA] flex items-center justify-center p-6">
        <div className="max-w-lg w-full bg-white rounded-[32px] p-12 text-center border border-gray-200 shadow-xl">
          <div className="w-24 h-24 bg-emerald-50 text-emerald-500 rounded-full flex items-center justify-center mx-auto mb-8 shadow-inner">
            <CheckCircle2 size={48} />
          </div>
          <h2 className="text-3xl font-serif text-[#101D42] mb-4">Dossier transmis</h2>
          <p className="text-[#9CA3AF] text-sm leading-relaxed mb-10">
            Votre dossier a bien été enregistré en toute sécurité. Il est actuellement en cours d’analyse par notre comité de sélection. Vous recevrez une notification prochainement.
          </p>
          <Link href="/login" className="inline-block w-full bg-[#101D42] hover:bg-[#1a2b5e] text-white font-semibold py-4 rounded-full transition-colors tracking-wide">
            RETOURNER À L'ACCUEIL
          </Link>
        </div>
      </div>
    );
  }

  return (
    
<div className="min-h-screen bg-[#FAFAFA] pb-12"> 
  

        <div className="max-w-3xl mx-auto"> </div>


      
      {/* NOUVEAU : HEADER DE DÉCONNEXION */}
      <header className="w-full bg-white border-b border-gray-200 px-6 py-4 flex items-center justify-between sticky top-0 z-50 shadow-sm">
        <div className="flex items-center gap-3">
          <div className="w-8 h-8 bg-[#101D42] rounded flex items-center justify-center shadow-sm">
            <span className="text-white font-serif font-bold text-lg">T</span>
          </div>
          <span className="text-xl font-semibold tracking-tight text-[#101D42] hidden sm:block">TechStratix</span>
        </div>
        
        <button 
          onClick={handleSignOut}
          className="flex items-center gap-2 text-sm font-medium text-gray-500 hover:text-red-600 transition-colors bg-gray-50 hover:bg-red-50 px-4 py-2 rounded-full"
        >
          <LogOut size={16} />
          <span className="hidden sm:inline">Quitter et reprendre plus tard</span>
        </button>
      </header>

      <div className="max-w-3xl mx-auto pt-12 px-4 sm:px-6">
        
        {/* HEADER DYNAMIQUE (Déjà existant) */}
        <div className="text-center mb-12"></div>
        {/* HEADER DYNAMIQUE */}
        <div className="text-center mb-12">
          <div className={`w-16 h-16 rounded-2xl flex items-center justify-center mx-auto mb-6 shadow-lg ${userRole === 'startup' ? 'bg-[#101D42] text-[#2563EB]' : userRole === 'mentor' ? 'bg-emerald-100 text-emerald-600' : 'bg-orange-100 text-orange-600'}`}>
            {userRole === "startup" && <Rocket size={32} />}
            {userRole === "mentor" && <Users size={32} />}
            {userRole === "investor" && <TrendingUp size={32} />}
          </div>
          <h1 className="text-3xl font-serif text-[#101D42] mb-3">Constitution du Dossier</h1>
          <p className="text-[#9CA3AF] font-medium">Étape finale pour accéder à l'écosystème TechStratix.</p>
        </div>

        <form onSubmit={handleSubmit} className="space-y-8">
          
          {/* ======================================================== */}
          {/* FORMULAIRE STARTUP                                       */}
          {/* ======================================================== */}
          {userRole === "startup" && (
            <>
              <BorderGlow backgroundColor="#ffffff" className="p-8">
                <div className="flex items-center gap-3 mb-8 border-b border-[#101D42]/10 pb-4">
                  <Building2 className="text-[#2563EB]" size={20} />
                  <h2 className="text-xl font-medium text-[#101D42]">Identité de l'entreprise</h2>
                </div>
                <div className="space-y-6">
                  <div>
                    <label className="block text-[10px] font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Nom de la startup <span className="text-red-500">*</span></label>
                    <input type="text" required value={formData.name} onChange={(e) => setFormData({...formData, name: e.target.value})} className="w-full px-4 py-3 rounded-xl border border-[#101D42]/15 shadow-sm focus:outline-none focus:border-[#2563EB] text-sm bg-white" placeholder="Ex: TechStratix Solutions" />
                  </div>
                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-6">

                      <label className="block text-[10px] font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Secteur d'activité <span className="text-red-500">*</span></label>
                      <select required value={formData.sector} onChange={(e) => setFormData({...formData, sector: e.target.value})} className="w-full px-4 py-3 rounded-xl border border-[#101D42]/15 shadow-sm focus:outline-none focus:border-[#2563EB] text-sm bg-white">
                        <option value="" disabled>Sélectionnez un secteur</option>

                  <h2 className="text-xl font-medium text-[#101D42]">Le Projet & Documents</h2>

"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { Rocket, Users, TrendingUp, Loader2, ArrowLeft } from "lucide-react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import { motion, AnimatePresence } from "framer-motion";

const SLIDER_MESSAGES = [
  {
    title: "Accélérez votre croissance.",
    desc: "Un écosystème conçu pour structurer votre projet, monter en compétences et réussir votre levée de fonds.",
    icon: <Rocket size={32} className="text-[#2563EB]" />
  },
  {
    title: "Transmettez votre expertise.",
    desc: "Accompagnez des fondateurs ambitieux et participez activement à l'innovation de demain.",
    icon: <Users size={32} className="text-emerald-400" />
  },
  {
    title: "Découvrez les pépites tech.",
    desc: "Un deal flow exclusif, qualifié et organisé pour vos prochains investissements stratégiques.",
    icon: <TrendingUp size={32} className="text-orange-400" />
  }
];

export default function RegisterPage() {
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");
  const [fullName, setFullName] = useState("");
  const [selectedRole, setSelectedRole] = useState("startup");
  
  const [loading, setLoading] = useState(false);
  const [errorMsg, setErrorMsg] = useState("");

  const [currentIndex, setCurrentIndex] = useState(0);
  useEffect(() => {
    const timer = setInterval(() => {
      setCurrentIndex((prev) => (prev + 1) % SLIDER_MESSAGES.length);
    }, 5000);
    return () => clearInterval(timer);
  }, []);

  const handleRegister = async (e: React.FormEvent) => {
    e.preventDefault();
    setErrorMsg("");

    if (password !== confirmPassword) {
      setErrorMsg("Les mots de passe ne correspondent pas.");
      return;
    }
    if (!fullName.trim()) {
      setErrorMsg("Veuillez saisir votre nom complet.");
      return;
    }

    setLoading(true);

    try {
      const { error } = await supabase.auth.signUp({
        email,
        password,
        options: { data: { role: selectedRole, full_name: fullName } }
      });
      if (error) throw error;
      router.push("/onboarding");
    } catch (error: any) {
      setErrorMsg(error.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen flex bg-[#FAFAFA]">
      <div className="w-full lg:w-1/2 flex flex-col justify-center items-center p-8 overflow-y-auto">
        <div className="w-full max-w-md">
          
          <Link href="/" className="inline-flex items-center gap-2 text-sm text-[#9CA3AF] hover:text-[#101D42] mb-10 transition-colors">
            <ArrowLeft size={16} /> Retour à l'accueil
          </Link>

          <h1 className="text-3xl font-medium tracking-tight text-[#101D42] mb-2">Rejoindre TechStratix</h1>
          <p className="text-[#9CA3AF] text-sm mb-8">Sélectionnez votre profil pour commencer.</p>

          <form onSubmit={handleRegister} className="space-y-6">
            <div className="grid grid-cols-3 gap-3">
              <div onClick={() => setSelectedRole("startup")} className={`cursor-pointer border rounded-2xl p-4 flex flex-col items-center gap-3 transition-all ${selectedRole === "startup" ? "border-[#2563EB] bg-[#EFF6FF] text-[#2563EB] shadow-sm" : "border-gray-200 text-gray-400 hover:border-[#101D42]"}`}>
                <Rocket size={24} />
                <span className="text-[10px] font-bold uppercase tracking-wider">Startup</span>
              </div>
              <div onClick={() => setSelectedRole("mentor")} className={`cursor-pointer border rounded-2xl p-4 flex flex-col items-center gap-3 transition-all ${selectedRole === "mentor" ? "border-emerald-500 bg-emerald-50 text-emerald-600 shadow-sm" : "border-gray-200 text-gray-400 hover:border-emerald-500"}`}>
                <Users size={24} />
                <span className="text-[10px] font-bold uppercase tracking-wider">Mentor</span>
              </div>
              <div onClick={() => setSelectedRole("investor")} className={`cursor-pointer border rounded-2xl p-4 flex flex-col items-center gap-3 transition-all ${selectedRole === "investor" ? "border-orange-500 bg-orange-50 text-orange-600 shadow-sm" : "border-gray-200 text-gray-400 hover:border-orange-500"}`}>
                <TrendingUp size={24} />
                <span className="text-[10px] font-bold uppercase tracking-wider">Investisseur</span>
              </div>
            </div>

            <div className="space-y-4">
              <div>
                <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Nom Complet</label>
                <input type="text" required value={fullName} onChange={(e) => setFullName(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="Jean Dupont" />
              </div>
              <div>
                <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">E-mail</label>
                <input type="email" required value={email} onChange={(e) => setEmail(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="nom@exemple.com" />
              </div>
              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Mot de passe</label>
                  <input type="password" required value={password} onChange={(e) => setPassword(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="••••••••" />
                </div>
                <div>
                  <label className="block text-xs font-bold tracking-[0.1em] text-[#9CA3AF] uppercase mb-2">Confirmer</label>
                  <input type="password" required value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} className="w-full px-4 py-3 rounded-xl border border-gray-200 focus:outline-none focus:border-[#2563EB] text-sm text-[#101D42]" placeholder="••••••••" />
                </div>
              </div>
            </div>

            {errorMsg && <div className="p-4 rounded-xl bg-red-50 border border-red-100 text-sm font-medium text-red-600">{errorMsg}</div>}

            <button type="submit" disabled={loading} className="w-full bg-[#101D42] hover:bg-[#1a2b5e] text-white font-semibold py-4 rounded-full tracking-wide transition-colors text-sm disabled:opacity-70 mt-4">
              {loading ? <Loader2 className="animate-spin mx-auto" size={20} /> : "CRÉER MON COMPTE"}
            </button>
          </form>
          
          <div className="mt-8 text-center text-sm">
            <span className="text-[#9CA3AF]">Vous avez déjà un compte ?</span>{" "}
            <Link href="/login" className="text-[#101D42] font-semibold hover:underline">Se connecter</Link>
          </div>
        </div>
      </div>

      {/* DROITE : LE PANNEAU MARKETING ANIMÉ */}
      <div className="hidden lg:flex w-1/2 bg-[#060B14] p-16 flex-col justify-between relative overflow-hidden">
        <div className="absolute top-[-10%] right-[-10%] w-[500px] h-[500px] bg-[#2563EB] opacity-20 rounded-full blur-[120px] pointer-events-none"></div>
        <div className="absolute bottom-[-10%] left-[-10%] w-[400px] h-[400px] bg-emerald-500 opacity-10 rounded-full blur-[100px] pointer-events-none"></div>

        <div className="relative z-10">
          <Link href="/" className="flex items-center gap-3">
            <div className="w-10 h-10 bg-white/5 backdrop-blur-md border border-white/10 rounded-xl flex items-center justify-center shadow-lg">
              <span className="text-white font-serif font-bold text-xl">T</span>
            </div>
            <span className="text-2xl font-medium tracking-tight text-white">TechStratix</span>
          </Link>
        </div>

        <div className="relative z-10 h-64 flex flex-col justify-center">
          <AnimatePresence mode="wait">
            <motion.div
              key={currentIndex}
              initial={{ opacity: 0, y: 20 }}
              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: -20 }}
              transition={{ duration: 0.5, ease: "easeOut" }}
            >
              <div className="mb-6 bg-white/5 w-16 h-16 rounded-2xl flex items-center justify-center border border-white/10 backdrop-blur-sm shadow-xl">
                {SLIDER_MESSAGES[currentIndex].icon}
              </div>
              <h2 className="text-4xl lg:text-5xl font-medium text-white mb-6 leading-[1.1] tracking-tight">
                {SLIDER_MESSAGES[currentIndex].title}
              </h2>
              <p className="text-gray-400 text-lg max-w-md leading-relaxed">
                {SLIDER_MESSAGES[currentIndex].desc}
              </p>
            </motion.div>
          </AnimatePresence>
        </div>

        <div className="relative z-10 flex gap-3 w-full max-w-sm">
          {SLIDER_MESSAGES.map((_, i) => (
            <div key={i} className="h-1 flex-1 bg-white/10 rounded-full overflow-hidden">
              {i === currentIndex && (
                <motion.div 
                  initial={{ width: 0 }} 
                  animate={{ width: "100%" }} 
                  transition={{ duration: 5, ease: "linear" }}
                  className="h-full bg-[#2563EB]"
                />
              )}
              {i < currentIndex && <div className="h-full bg-[#2563EB]" />}
            </div>
          ))}
        </div>

      </div>
    </div>
  );
}
