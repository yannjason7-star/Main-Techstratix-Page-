"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { Loader2, Rocket, Briefcase, TrendingUp, Users, Calendar, ArrowRight, Target, CheckCircle2 } from "lucide-react";
import Link from "next/link";

export default function DashboardPage() {
  const [role, setRole] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);
  
  // États pour stocker les vraies données
  const [startupStats, setStartupStats] = useState({ progress: 0, completedTasks: 0, totalTasks: 0 });
  const [networkStats, setNetworkStats] = useState({ activeStartups: 0, totalEvents: 0 });
  const [dealFlow, setDealFlow] = useState<any[]>([]);

  useEffect(() => {
    const fetchDashboardData = async () => {
      try {
        const { data: { user } } = await supabase.auth.getUser();
        if (!user) return;

        // 1. Quel est mon rôle ?
        const { data: profile } = await supabase.from("profiles").select("role").eq("id", user.id).single();
        if (!profile) return;
        setRole(profile.role);

        // 2. Récupérer les données selon le rôle
        if (profile.role === 'startup') {
          // Calcul de la VRAIE progression basée sur les tâches
          const { data: tasks } = await supabase.from('tasks').select('status').eq('user_id', user.id);
          if (tasks && tasks.length > 0) {
            const completed = tasks.filter(t => t.status === 'done').length;
            const progressPercent = Math.round((completed / tasks.length) * 100);
            setStartupStats({ progress: progressPercent, completedTasks: completed, totalTasks: tasks.length });
          }
        } 
        else if (profile.role === 'mentor') {
          // Compter les vraies startups actives
          const { count: startupCount } = await supabase.from('profiles').select('*', { count: 'exact', head: true }).eq('role', 'startup').eq('status', 'approved');
          const { count: eventCount } = await supabase.from('events').select('*', { count: 'exact', head: true });
          setNetworkStats({ activeStartups: startupCount || 0, totalEvents: eventCount || 0 });
        }
        else if (profile.role === 'investor') {
          // Afficher les 3 dernières startups du Deal Flow
          const { data: startups } = await supabase
            .from('profiles')
            .select('id, full_name, startup_profiles(sector, startup_name)')
            .eq('role', 'startup')
            .eq('status', 'approved')
            .order('created_at', { ascending: false })
            .limit(3);
          
          if (startups) setDealFlow(startups);
        }

      } catch (error) {
        console.error("Erreur Dashboard:", error);
      } finally {
        setLoading(false);
      }
    };

    fetchDashboardData();
  }, []);

  if (loading) {
    return <div className="h-[50vh] flex items-center justify-center text-[#2563EB]"><Loader2 className="animate-spin" size={40} /></div>;
  }

  // ==========================================
  // VUE : STARTUP
  // ==========================================
  if (role === "startup") {
    return (
      <div className="max-w-5xl mx-auto pb-12">
        <div className="mb-10">
          <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Vue d'ensemble</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Ma Progression</h2>
        </div>
        
        <div className="bg-white rounded-[32px] border border-gray-200 p-10 shadow-sm relative overflow-hidden">
          <div className="absolute top-0 right-0 w-64 h-64 bg-[#2563EB] opacity-5 rounded-full blur-[80px]"></div>
          
          <div className="flex items-center gap-4 mb-6 relative z-10">
            <div className="w-14 h-14 bg-blue-50 text-[#2563EB] rounded-2xl flex items-center justify-center shadow-inner">
              <Rocket size={24} />
            </div>
            <div>
              <h3 className="text-xl font-medium text-[#101D42]">Bienvenue dans votre espace d'accélération.</h3>
              <p className="text-[#9CA3AF] text-sm">Utilisez le menu pour suivre vos objectifs et contacter vos mentors.</p>
            </div>
          </div>

          <div className="bg-[#FAFAFA] p-6 rounded-2xl border border-gray-100 relative z-10">
            <div className="flex justify-between items-end mb-3">
              <div className="flex items-center gap-2 text-[#101D42] font-medium text-sm">
                <Target size={18} className="text-[#2563EB]" /> Feuille de route
              </div>
              <p className="text-2xl font-bold text-[#101D42]">{startupStats.progress}%</p>
            </div>
            <div className="h-3 w-full bg-gray-200 rounded-full overflow-hidden mb-3">
              <div className="h-full bg-[#2563EB] transition-all duration-1000" style={{ width: `${startupStats.progress}%` }}></div>
            </div>
            <p className="text-xs font-bold text-[#9CA3AF] uppercase tracking-wider">
              {startupStats.completedTasks} tâches terminées sur {startupStats.totalTasks}
            </p>
          </div>
        </div>
      </div>
    );
  }

  // ==========================================
  // VUE : MENTOR
  // ==========================================
  if (role === "mentor") {
    return (
      <div className="max-w-6xl mx-auto pb-12">
        <div className="mb-10">
          <p className="text-[10px] font-bold tracking-[0.15em] text-emerald-600 uppercase mb-2">Espace Expert</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Bienvenue, Mentor.</h2>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-6 mb-8">
          <div className="bg-[#101D42] text-white p-8 rounded-[32px] shadow-sm relative overflow-hidden group">
            <div className="absolute top-[-20%] right-[-10%] w-40 h-40 bg-[#2563EB] opacity-20 blur-[40px] group-hover:opacity-40 transition-opacity"></div>
            <Briefcase size={28} className="mb-6 text-[#2563EB]" />
            <p className="text-xs font-bold tracking-widest text-gray-400 uppercase mb-2">Startups dans l'incubateur</p>
            <h3 className="text-5xl font-serif">{networkStats.activeStartups}</h3>
          </div>
          
          <div className="bg-white border border-gray-200 p-8 rounded-[32px] shadow-sm hover:border-emerald-500/30 transition-colors">
            <Calendar size={28} className="mb-6 text-emerald-500" />
            <p className="text-xs font-bold tracking-widest text-[#9CA3AF] uppercase mb-2">Événements programmés</p>
            <h3 className="text-5xl font-serif text-[#101D42]">{networkStats.totalEvents}</h3>
          </div>
        </div>
      </div>
    );
  }

  // ==========================================
  // VUE : INVESTISSEUR
  // ==========================================
  if (role === "investor") {
    return (
      <div className="max-w-6xl mx-auto pb-12">
        <div className="mb-10 flex justify-between items-end">
          <div>
            <p className="text-[10px] font-bold tracking-[0.15em] text-orange-600 uppercase mb-2">Sourcing & Opportunités</p>
            <h2 className="text-2xl font-medium text-[#101D42]">Deal Flow Récent</h2>
          </div>
          <Link href="/reseau" className="text-[10px] font-bold text-[#2563EB] uppercase hover:underline">Voir tout l'annuaire</Link>
        </div>

        {dealFlow.length === 0 ? (
          <div className="bg-white p-12 rounded-[32px] border border-gray-200 text-center shadow-sm">
            <TrendingUp size={40} className="mx-auto text-gray-300 mb-4" />
            <p className="text-[#9CA3AF] font-medium">Aucune nouvelle startup n'a été approuvée récemment.</p>
          </div>
        ) : (
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
            {dealFlow.map((startup, i) => {
              const details = startup.startup_profiles?.[0] || {};
              const sName = details.startup_name || startup.full_name;
              const sSector = details.sector || "Non spécifié";
              
              return (
                <div key={i} className="bg-white rounded-[24px] border border-gray-200 p-6 shadow-sm hover:border-[#2563EB]/40 transition-colors flex flex-col justify-between">
                  <div>
                    <div className="w-12 h-12 rounded-xl bg-[#F8FAFC] border border-gray-100 flex items-center justify-center text-xl font-serif text-[#101D42] mb-4">
                      {sName.charAt(0)}
                    </div>
                    <h4 className="text-lg font-medium text-[#101D42] mb-1 truncate">{sName}</h4>
                    <span className="text-[10px] font-bold tracking-[0.1em] text-[#9CA3AF] uppercase bg-gray-50 px-2 py-1 rounded">
                      {sSector}
                    </span>
                  </div>
                  <Link href="/reseau" className="mt-6 w-full flex items-center justify-center gap-2 bg-[#F8FAFC] hover:bg-[#EFF6FF] text-[#2563EB] py-3 rounded-full text-xs font-bold transition-colors">
                    VOIR LE PROFIL <ArrowRight size={14} />
                  </Link>
                </div>
              );
            })}
          </div>
        )}
      </div>
    );
  }

  return null;
}

"use client";

import { useState } from "react";
import { FileText, Download, Upload, MoreVertical, Loader2, FileCheck2, Trash2 } from "lucide-react";

// Fausses données pour le prototype
const INITIAL_DOCS = [
  { id: 1, name: "Pitch_Deck_GreenTech_V3.pdf", size: "4.2 MB", date: "Aujourd'hui, 10:30", type: "pdf", uploader: "Alexandre" },
  { id: 2, name: "Previsionnel_Financier_2024.xlsx", size: "1.1 MB", date: "12 Octobre", type: "excel", uploader: "Sarah (Mentor)" },
  { id: 3, name: "Kbis_A_Jour.pdf", size: "0.8 MB", date: "05 Octobre", type: "pdf", uploader: "Alexandre" },
];

export default function DocsPage() {
  const [docs, setDocs] = useState(INITIAL_DOCS);
  const [isUploading, setIsUploading] = useState(false);
  const [downloadingId, setDownloadingId] = useState<number | null>(null);

  // Fonction pour simuler l'import d'un document
  const handleUpload = async () => {
    try {
      setIsUploading(true);
      
      // Simulation d'un upload de 2 secondes
      await new Promise(resolve => setTimeout(resolve, 2000));
      
      const newDoc = {
        id: Date.now(),
        name: "Nouveau_Document_Signe.pdf",
        size: "2.5 MB",
        date: "À l'instant",
        type: "pdf",
        uploader: "Alexandre"
      };
      
      setDocs([newDoc, ...docs]);
      alert("✅ Document importé avec succès dans votre espace sécurisé.");
    } catch (error: any) {
      alert("❌ Erreur lors de l'import : " + error.message);
    } finally {
      setIsUploading(false);
    }
  };

  // Fonction pour simuler le téléchargement
  const handleDownload = async (id: number, name: string) => {
    try {
      setDownloadingId(id);
      
      // Simulation temps de téléchargement
      await new Promise(resolve => setTimeout(resolve, 1500));
      
      alert(`✅ Téléchargement de "${name}" terminé !`);
    } catch (error: any) {
      alert("❌ Échec du téléchargement : " + error.message);
    } finally {
      setDownloadingId(null);
    }
  };

  const handleDelete = (id: number) => {
    if (confirm("Êtes-vous sûr de vouloir supprimer ce document ?")) {
      setDocs(docs.filter(d => d.id !== id));
    }
  };

  return (
    <div className="max-w-5xl mx-auto h-full flex flex-col">
      
      <div className="flex items-center justify-between mb-8">
        <div>
          <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Espace sécurisé</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Documents & Data Room</h2>
        </div>
        
        <button 
          onClick={handleUpload}
          disabled={isUploading}
          className="flex items-center gap-2 bg-[#101D42] hover:bg-[#1a2b5e] text-white px-6 py-3 rounded-full text-xs font-semibold tracking-wide transition-all shadow-sm disabled:opacity-70"
        >
          {isUploading ? <Loader2 size={16} className="animate-spin" /> : <Upload size={16} />}
          {isUploading ? "IMPORTATION..." : "IMPORTER UN FICHIER"}
        </button>
      </div>

      <div className="bg-white rounded-[24px] border border-gray-200 shadow-sm flex-1 overflow-hidden flex flex-col">
        {/* En-tête du tableau (Léger et discret) */}
        <div className="grid grid-cols-12 gap-4 p-6 border-b border-gray-100 bg-[#FAFAFA] text-[10px] font-bold tracking-[0.1em] text-[#9CA3AF] uppercase">
          <div className="col-span-6 md:col-span-5">Nom du fichier</div>
          <div className="col-span-3 hidden md:block">Ajouté par</div>
          <div className="col-span-4 md:col-span-3">Date d'ajout</div>
          <div className="col-span-2 md:col-span-1 text-right">Taille</div>
        </div>

        {/* Liste des fichiers */}
        <div className="flex-1 overflow-y-auto p-2">

              <FileCheck2 size={48} className="mx-auto text-gray-200 mb-4" />
              <p className="text-[#9CA3AF] font-medium">Votre espace documentaire est vide.</p>
            </div>
          ) : (
            docs.map((doc) => (
              <div key={doc.id} className="grid grid-cols-12 gap-4 p-4 items-center hover:bg-[#F8FAFC] rounded-2xl transition-colors group border-b border-gray-50 last:border-0">

                  <div className="w-10 h-10 bg-[#EFF6FF] text-[#2563EB] rounded-xl flex items-center justify-center shrink-0">

                  </div>

                </div>

                {/* Auteur (Caché sur mobile) */}
                <div className="col-span-3 hidden md:flex items-center">

                  </span>
                </div>

                {/* Date */}
                <div className="col-span-4 md:col-span-3 flex items-center">

    try {
      setLoading(true);

      if (data) setCourses(data);
    } catch (error: any) {
      alert("❌ Impossible de charger les formations : " + error.message);
      console.error(error);
    } finally {
      setLoading(false);
    }

    try {
      setActionLoading(courseId);
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) throw new Error("Vous devez être connecté.");

      await new Promise(resolve => setTimeout(resolve, 800)); // Simulation de chargement

      if (isPremium) {
        alert(`💳 Redirection vers la passerelle de paiement pour régler ${price}€ pour "${courseTitle}".`);
      } else {
        alert(`✅ Succès ! Vous êtes maintenant inscrit au cours "${courseTitle}".`);
        setSelectedCourse(null); // On ferme la fenêtre après l'inscription

      alert("❌ Erreur : " + error.message);

      setActionLoading(null);

"use client";

import { useState, useEffect } from "react";
import { supabase } from "@/lib/supabase";
import { Target, CheckCircle2, Circle, Plus, Loader2, Trash2 } from "lucide-react";

export default function ProjetPage() {
  const [tasks, setTasks] = useState<any[]>([]);
  const [newTaskTitle, setNewTaskTitle] = useState("");
  const [loading, setLoading] = useState(true);
  const [adding, setAdding] = useState(false);
  const [userId, setUserId] = useState<string | null>(null);

  useEffect(() => {
    fetchSessionAndTasks();
  }, []);

  const fetchSessionAndTasks = async () => {
    setLoading(true);
    // Utilisation de getUser() qui est beaucoup plus fiable et sécurisé que getSession()
    const { data: { user }, error: userError } = await supabase.auth.getUser();
    
    if (user) {
      setUserId(user.id);
      
      const { data, error } = await supabase
        .from("tasks")
        .select("*")
        .order("created_at", { ascending: false });

      if (error) {
        alert("Erreur de lecture : " + error.message);
      }
      if (data) setTasks(data);
    } else {
      console.log("Utilisateur non détecté.");
    }
    setLoading(false);
  };

  const handleAddTask = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newTaskTitle.trim()) return;
    
    // Si l'utilisateur n'est pas détecté, on le dit !
    if (!userId) {
      alert("Erreur : Votre session est introuvable. Essayez de vous déconnecter et vous reconnecter.");
      return;
    }

    setAdding(true);
    
    const { data, error } = await supabase
      .from("tasks")
      .insert([
        { 
          user_id: userId, 
          title: newTaskTitle, 
          status: "todo" 
        }
      ])
      .select();

    // Si Supabase bloque l'action, on affiche l'erreur exacte
    if (error) {
      alert("Erreur Supabase : " + error.message);
      console.error(error);
    }

    if (data && data.length > 0) {
      setTasks([data[0], ...tasks]);
      setNewTaskTitle("");
    }
    
    setAdding(false);
  };

  const toggleTaskStatus = async (id: string, currentStatus: string) => {
    const newStatus = currentStatus === "done" ? "todo" : "done";
    setTasks(tasks.map(t => t.id === id ? { ...t, status: newStatus } : t));

    const { error } = await supabase
      .from("tasks")
      .update({ status: newStatus })
      .eq("id", id);
      
    if (error) alert("Erreur mise à jour : " + error.message);
  };

  const deleteTask = async (id: string) => {
    setTasks(tasks.filter(t => t.id !== id));
    const { error } = await supabase.from("tasks").delete().eq("id", id);
    if (error) alert("Erreur suppression : " + error.message);
  };

  return (
    <div className="max-w-4xl bg-white rounded-[24px] border border-gray-200 p-10 shadow-sm">
      <div className="flex items-center justify-between mb-8">
        <div>
          <p className="text-[10px] font-bold tracking-[0.15em] text-[#9CA3AF] uppercase mb-2">Feuille de route</p>
          <h2 className="text-2xl font-medium text-[#101D42]">Mes Objectifs</h2>
        </div>
      </div>

      <form onSubmit={handleAddTask} className="mb-10 flex gap-4">
        <input 
          type="text" 
          value={newTaskTitle}
          onChange={(e) => setNewTaskTitle(e.target.value)}
          placeholder="Nouvelle étape pour ma startup..." 
          className="flex-1 border border-gray-200 rounded-full px-6 py-3 text-sm focus:outline-none focus:border-[#101D42] text-[#101D42]"
          disabled={adding}
        />
        <button 
          type="submit" 
          disabled={adding || !newTaskTitle.trim()}
          className="bg-[#101D42] text-white px-6 py-3 rounded-full text-xs font-semibold tracking-wide flex items-center gap-2 disabled:opacity-50 hover:bg-[#1a2b5e] transition-colors"
        >
          {adding ? <Loader2 size={16} className="animate-spin" /> : <Plus size={16} />} 
          AJOUTER
        </button>
      </form>

      {loading ? (
        <div className="flex justify-center items-center py-20 text-[#9CA3AF]">
          <Loader2 className="animate-spin" size={32} />
        </div>
      ) : tasks.length === 0 ? (
        <div className="text-center py-16 border-2 border-dashed border-gray-100 rounded-2xl">
          <Target size={48} className="mx-auto text-gray-200 mb-4" />
          <p className="text-[#9CA3AF] font-medium">Aucun objectif pour le moment.</p>
        </div>
      ) : (
        <div className="space-y-4">
          {tasks.map((task) => {
            const isDone = task.status === "done";
            return (
              <div key={task.id} className="flex gap-4 items-center group bg-white border border-gray-100 hover:border-gray-200 p-5 rounded-2xl transition-all">
                <button onClick={() => toggleTaskStatus(task.id, task.status)} className="mt-0.5 flex-shrink-0">
                  {isDone ? <CheckCircle2 className="text-[#2563EB]" size={24} /> : <Circle className="text-gray-300 hover:text-[#101D42] transition-colors" size={24} />}
                </button>
                <div className="flex-1">
                  <h3 className={`text-base font-medium transition-colors ${isDone ? 'text-[#9CA3AF] line-through' : 'text-[#101D42]'}`}>
                    {task.title}
                  </h3>
                </div>
                <button 
                  onClick={() => deleteTask(task.id)}
                  className="opacity-0 group-hover:opacity-100 text-gray-300 hover:text-red-500 transition-all p-2"
                >
                  <Trash2 size={18} />
                </button>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
